📌 Part 2: Express 프레임워크Express 5.x

2.1 Express란?

Express는 Node.js를 위한 빠르고 유연한 웹 프레임워크입니다.

  • ✅ 간결한 API
  • ✅ 미들웨어 아키텍처
  • ✅ 라우팅 시스템
  • ✅ RESTful API 구축에 최적

2.2 프로젝트 초기화

terminal
# 프로젝트 폴더 생성
mkdir my-express-app
cd my-express-app

# package.json 생성
npm init -y

# 필수 패키지 설치
npm install express cors dotenv

# 개발 도구 설치
npm install --save-dev nodemon

📌 Part 3: RESTful API 설계REST 원칙

3.1 REST란?

REST (Representational State Transfer)는 웹 서비스 설계 아키텍처입니다.

3.2 HTTP 메서드

메서드 용도 예제 CRUD
GET 데이터 조회 GET /api/posts Read
POST 데이터 생성 POST /api/posts Create
PUT 데이터 전체 수정 PUT /api/posts/1 Update
PATCH 데이터 부분 수정 PATCH /api/posts/1 Update
DELETE 데이터 삭제 DELETE /api/posts/1 Delete

3.3 RESTful 엔드포인트 설계

API 설계 예제
// 좋은 RESTful API 설계
GET    /api/posts          // 모든 포스트 조회
GET    /api/posts/:id      // 특정 포스트 조회
POST   /api/posts          // 새 포스트 생성
PUT    /api/posts/:id      // 포스트 전체 수정
PATCH  /api/posts/:id      // 포스트 부분 수정
DELETE /api/posts/:id      // 포스트 삭제

// 중첩 리소스
GET    /api/posts/:id/comments    // 특정 포스트의 댓글
POST   /api/posts/:id/comments    // 특정 포스트에 댓글 작성

📌 Part 4: 미들웨어 (Middleware)핵심 개념

4.1 미들웨어란?

요청(Request)과 응답(Response) 사이에서 실행되는 함수입니다.

middleware.js
// 기본 미들웨어 구조
const myMiddleware = (req, res, next) => {
    console.log('요청 처리 중...');
    next(); // 다음 미들웨어로 이동
};

// 로깅 미들웨어
const logger = (req, res, next) => {
    const timestamp = new Date().toISOString();
    console.log(`[${timestamp}] ${req.method} ${req.url}`);
    next();
};

// 인증 미들웨어
const authenticate = (req, res, next) => {
    const token = req.headers.authorization;
    
    if (!token) {
        return res.status(401).json({ error: '인증 필요' });
    }
    
    // 토큰 검증 로직...
    next();
};

// 에러 핸들링 미들웨어
const errorHandler = (err, req, res, next) => {
    console.error(err.stack);
    res.status(500).json({ 
        error: '서버 오류',
        message: err.message 
    });
};

4.2 내장 미들웨어

built-in-middleware.js
import express from 'express';
const app = express();

// JSON 파싱

// URL-encoded 데이터 파싱

4.3 서드파티 미들웨어

third-party-middleware.js

        const posts = await Post.find();

        res.status(500).json({ error: error.message });

// 비동기 함수 체이닝
app.post('/api/posts', async (req, res) => {
    try {
        const post = await Post.create(req.body);

    } catch (error) {
        res.status(400).json({ error: error.message });
    }
});

5.4 데이터 검증 (Joi, Zod)

validation.js
import Joi from 'joi';

// 포스트 스키마
const postSchema = Joi.object({
    title: Joi.string().min(3).max(100).required(),
    content: Joi.string().min(10).required(),
    author: Joi.string().required(),
    tags: Joi.array().items(Joi.string())
});

// 검증 미들웨어
const validatePost = (req, res, next) => {
    const { error } = postSchema.validate(req.body);
    
    if (error) {
        return res.status(400).json({ 
            error: error.details[0].message 
        });
    }
    
    next();
};

// 라우트에 적용
app.post('/api/posts', validatePost, async (req, res) => {
    // 검증된 데이터 처리...
});

📌 Part 6: 프론트엔드 연동Fetch API

6.1 최신 Fetch API

frontend.js
// GET 요청
async function getPosts() {
    try {
        const response = await fetch('http://localhost:3000/api/posts');
        
        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }
        
        const posts = await response.json();
        console.log(posts);
    } catch (error) {
        console.error('Error:', error);
    }
}

// POST 요청
async function createPost(postData) {
    try {
        const response = await fetch('http://localhost:3000/api/posts', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
            },
            body: JSON.stringify(postData)
        });
        
        const newPost = await response.json();
        return newPost;
    } catch (error) {
        console.error('Error:', error);
    }
}

// 사용 예제
const newPost = await createPost({
    title: '새 포스트',
    content: '내용입니다.',
    author: '홍길동'
});
